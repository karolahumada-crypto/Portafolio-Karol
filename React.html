const nombres = productos.map(

02 — inventory.js
Reduce — Total
Calcular valor total del stock
.reduce()
Precio × cantidad por producto
Acumular con valor inicial 0
Obtener el total del inventario
inventory.js 
const productos = [
  { nombre: "Laptop",  precio: 1200, cantidad: 5  },
  { nombre: "Mouse",   precio: 25,   cantidad: 20 },
  { nombre: "Teclado", precio: 45,   cantidad: 15 }
];

const total = productos.reduce((acc, p) => {
  return acc + (p.precio * p.cantidad);
}, 0);
// output
03 — max_stock.js
Reduce — Mayor Stock
Producto con más unidades
.reduce()
Comparar cantidad en cada iteración
Retener el acumulador mayor
Sin valor inicial — usa el primer elemento
max_stock.js 
const productos = [
  { nombre: "Laptop",  cantidad: 5  },
  { nombre: "Mouse",   cantidad: 20 },
  { nombre: "Teclado", cantidad: 15 }
];

const mayor = productos.reduce((max, p) => {
  return p.cantidad > max.cantidad ? p : max;
});

// → { nombre: "Mouse", cantidad: 20 }
// output
